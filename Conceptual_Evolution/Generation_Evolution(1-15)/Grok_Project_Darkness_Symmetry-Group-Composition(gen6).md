# generate_instances.py - Generate Tseitin SAT on expanders
import networkx as nx
import numpy as np
import argparse
import pickle

def generate_expander(n, d=3):
    """Generate approx Ramanujan expander: random d-regular graph with high expansion."""
    G = nx.random_regular_graph(d, n)
    while not nx.is_connected(G):  # Ensure connected
        G = nx.random_regular_graph(d, n)
    return G

def tseitin_formula(G, charge_vertex=0):
    """Build 3-CNF Tseitin formula."""
    n = G.number_of_nodes()
    edges = list(G.edges())
    variables = {e: f"x_{i}" for i, e in enumerate(edges)}  # Vars per edge
    clauses = []
    charge = [0] * n
    charge[charge_vertex] = 1  # Odd charge at v*

    for v in range(n):
        incident = [variables[e] for e in edges if v in e]
        if len(incident) != 3: raise ValueError("Not 3-regular")
        a, b, c = incident
        parity = charge[v]
        if parity == 0:  # Even
            clauses.extend([
                (f"~{a}", f"~{b}", f"~{c}"),
                (f"~{a}", b, c),
                (a, f"~{b}", c),
                (a, b, f"~{c}")
            ])
        else:  # Odd
            clauses.extend([
                (a, b, c),
                (a, f"~{b}", f"~{c}"),
                (f"~{a}", b, f"~{c}"),
                (f"~{a}", f"~{b}", c)
            ])
    return clauses, variables

def primal_graph(clauses):
    """Build G_pr: vars connected if in same clause."""
    vars = set()
    for clause in clauses:
        vars.update([v.lstrip('~') for v in clause])
    G_pr = nx.Graph()
    G_pr.add_nodes_from(vars)
    for clause in clauses:
        clause_vars = [v.lstrip('~') for v in clause]
        for i in range(len(clause_vars)):
            for j in range(i+1, len(clause_vars)):
                G_pr.add_edge(clause_vars[i], clause_vars[j])
    return G_pr

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--n", type=int, default=50, help="Number of vertices")
    parser.add_argument("--q", type=int, default=None, help="Lift param (default n)")
    args = parser.parse_args()
    n = args.n
    q = args.q if args.q else n  # For lifted family φ_{G,q}

        pickle.dump((G, clauses, vars, G_pr), f)
    print(f"Generated instance for n={n}, q={q}, clauses={len(clauses)}")
